    SELECTCOLUMNS(
        FILTER(INFO.VIEW.COLUMNS(), CONTAINSSTRING([Expression], "SLA Total")),

    SELECTCOLUMNS(
        FILTER(INFO.VIEW.MEASURES(), CONTAINSSTRING([Expression], "SLA Total")),
        "Tipo", "Medida", "Tabela", [Table], "Nome", [Name])
)
