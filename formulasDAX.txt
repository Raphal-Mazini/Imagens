        let
            b = Binary.ToList(Binary.Range(bin, 0, 4))
        in
            if List.FirstN(b, 2) = {80, 75} then "OK - xlsx normal"
            else if b = {208, 207, 17, 224} then "PROBLEMA - formato antigo (.xls) ou arquivo protegido"
            else "PROBLEMA - nao e um Excel (" & Text.Combine(List.Transform(b, Text.From), ",") & ")",

    Urls = {
        "https://orizonbr.sharepoint.com/sites/IntelignciaOperacional/Documentos%20Compartilhados/BVP/Base%20de%20dados%20Colaboradores%20BVP.xlsx",

        "https://orizonbr.sharepoint.com/sites/PortaldaQualidadeOperacional/Bases%20de%20Monitoria/9.%20BVP/Base%20de%20Monitoria/Base%20de%20Monitoria%20BVP%20-%20Novo%20Modelo.xlsm",
        "https://orizonbr.sharepoint.com/sites/BibliotecaBVP/Documentos%20Compartilhados/Produ%C3%A7%C3%A3o/Cruzamentos%20di%C3%A1rio%202026.xlsx",
        "https://orizonbr.sharepoint.com/sites/BibliotecaBVP/Documentos%20Compartilhados/Relat%C3%B3rios%20BVP/Avisos%20encerrados%20Prescrito/Avisos%20Encerrados.xlsx",

        "https://orizonbr.sharepoint.com/sites/CMO/Documentos%20Compartilhados/Produtividade%20BVP.xlsx",
        "https://orizonbr.sharepoint.com/sites/IntelignciaOperacional/Documentos%20Compartilhados/BVP/Controle%20de%20Produtividade_BVP%20-%20Back%20Office%202023.xlsx",

        "https://orizonbr-my.sharepoint.com/personal/tpnazare_orizon_com_br/Documents/Arquivos%20de%20Chat%20do%20Microsoft%20Teams/AP%C3%93LICES%20CARDIF%203.xlsx",
        "https://orizonbr-my.sharepoint.com/personal/cbarros_orizon_com_br/Documents/Malote%20Previdencia%202026.xlsx"
    },
    Arquivos = Table.FromRecords(List.Transform(Urls, (u) =>
        [Arquivo = Uri.UnescapeDataString(u),
         Diagnostico = try Classificar(Web.Contents(u)) otherwise "ERRO AO ABRIR (sem acesso ou nao encontrado)"])),

    Site = SharePoint.Files("https://orizonbr.sharepoint.com/sites/IntelignciaOperacional/", [ApiVersion = 15]),
    PastasBVP = Table.SelectRows(Site, each List.Contains({
        "https://orizonbr.sharepoint.com/sites/IntelignciaOperacional/Documentos Compartilhados/BVP/Faturamento BVP/Faturamento SISE/",
        "https://orizonbr.sharepoint.com/sites/IntelignciaOperacional/Documentos Compartilhados/BVP/Faturamento BVP/Faturamento Portal/"},
        [Folder Path])),
    Ultimos = Table.Group(PastasBVP, {"Folder Path"}, {{"Ultimo", each Table.Max(_, "Date created")}}),
    Pastas = Table.FromRecords(List.Transform(Ultimos[Ultimo], (r) =>
        [Arquivo = r[Folder Path] & r[Name] & " (mais recente da pasta)",
         Diagnostico = try Classificar(r[Content]) otherwise "ERRO AO ABRIR"])),

    Resultado = Table.Combine({Arquivos, Pastas})
