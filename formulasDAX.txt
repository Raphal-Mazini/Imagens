EVALUATE
VAR t = FILTER('Gestão de Insumos (Main)',
    NOT ISBLANK('Gestão de Insumos (Main)'[Data_Conclusao_Qualificacao_Tecnica_Item])
    && NOT ISBLANK('Gestão de Insumos (Main)'[Data_Conclusao_Qualificacao_Analitica_Item]))
RETURN
ROW(
    "Itens concluidos", COUNTROWS(t),
    "Media Enfermeira pronta", AVERAGEX(t, 'Gestão de Insumos (Main)'[SLA_da_enfermeira]),
    "Media Tecnica calculada", AVERAGEX(t, 'Gestão de Insumos (Main)'[SLA QUALIFICAÇÃO Tec ITEM]),
    "Media Analitica pronta", AVERAGEX(t, 'Gestão de Insumos (Main)'[Sla_Qualificacao_Analitica]),
    "Media Analitica calculada", AVERAGEX(t, 'Gestão de Insumos (Main)'[SLA QUALIFICAÇÃO Analitc ITEM]),
    "Media Orizon pronta", AVERAGEX(t, 'Gestão de Insumos (Main)'[SLA_Orizon]),
    "Orizon diferente da soma", COUNTROWS(FILTER(t,
        ABS('Gestão de Insumos (Main)'[SLA_Orizon] - ('Gestão de Insumos (Main)'[SLA_da_enfermeira] + 'Gestão de Insumos (Main)'[Sla_Qualificacao_Analitica])) >= 0.01))
)
