EVALUATE
UNION(
    SELECTCOLUMNS(
        FILTER(INFO.PARTITIONS(),
            CONTAINSSTRING([QueryDefinition], "xls") || CONTAINSSTRING([QueryDefinition], "SharePoint")),
        "Origem", "Tabela",
        "Nome", [Name],
        "Codigo", [QueryDefinition]),
    SELECTCOLUMNS(
        FILTER(INFO.EXPRESSIONS(),
            CONTAINSSTRING([Expression], "xls") || CONTAINSSTRING([Expression], "SharePoint")),
        "Origem", "Consulta",
        "Nome", [Name],
        "Codigo", [Expression])
)
