EVALUATE
VAR t = 'Gestão de Insumos (Main)'
RETURN
ROW(
    "Itens", COUNTROWS(t),
    "Media Enfermeira pronta", AVERAGE('Gestão de Insumos (Main)'[SLA_da_enfermeira]),
    "Media Tecnica calculada", AVERAGE('Gestão de Insumos (Main)'[SLA QUALIFICAÇÃO Tec ITEM]),
    "Media Analitica pronta", AVERAGE('Gestão de Insumos (Main)'[Sla_Qualificacao_Analitica]),
    "Media Analitica calculada", AVERAGE('Gestão de Insumos (Main)'[SLA QUALIFICAÇÃO Analitc ITEM]),
    "Media SLA_Orizon pronta", AVERAGE('Gestão de Insumos (Main)'[SLA_Orizon]),
    "Enfermeira preenchida", COUNTROWS(FILTER(t, NOT ISBLANK('Gestão de Insumos (Main)'[SLA_da_enfermeira]))),
    "Enfermeira igual Tecnica", COUNTROWS(FILTER(t, NOT ISBLANK('Gestão de Insumos (Main)'[SLA_da_enfermeira])
        && ABS('Gestão de Insumos (Main)'[SLA_da_enfermeira] - 'Gestão de Insumos (Main)'[SLA QUALIFICAÇÃO Tec ITEM]) < 0.01)),
    "Orizon preenchida", COUNTROWS(FILTER(t, NOT ISBLANK('Gestão de Insumos (Main)'[SLA_Orizon]))),
    "Orizon igual Enf + Analitica", COUNTROWS(FILTER(t, NOT ISBLANK('Gestão de Insumos (Main)'[SLA_Orizon])
        && ABS('Gestão de Insumos (Main)'[SLA_Orizon] - ('Gestão de Insumos (Main)'[SLA_da_enfermeira] + 'Gestão de Insumos (Main)'[Sla_Qualificacao_Analitica])) < 0.01))
)
